### 1.2 详细实体关系

#### 1.2.1 SceneTemplate (场景模板) - 核心配置实体

```yaml
SceneTemplate:
  # 基础信息
  id: string                    # 模板唯一标识
  name: string                  # 显示名称
  description: string           # 描述
  version: string               # 版本
  type: SceneType               # 场景类型
  
  # 依赖配置
  dependencies:
    required: []                # 必需依赖
    optional: []                # 可选依赖
  
  # 角色配置
  roles:
    - roleName: string          # 角色名称
      displayName: string       # 显示名称
      permissions: []           # 权限列表
      capabilities: []          # 能力列表
  
  # 激活步骤配置
  activationSteps:
    MANAGER: []                 # 领导激活步骤
    EMPLOYEE: []                # 员工激活步骤
    STANDALONE: []              # 独立技能步骤
  
  # 菜单配置
  menus:
    - menuId: string
      displayName: string
      icon: string
      route: string
      requiredRole: string
  
  # UI技能配置
  uiSkills:
    - skillId: string
      displayName: string
      entryPoint: string
  
  # 私有能力配置
  privateCapabilities:
    - capabilityId: string
      configSchema: object
      defaultConfig: object
```
