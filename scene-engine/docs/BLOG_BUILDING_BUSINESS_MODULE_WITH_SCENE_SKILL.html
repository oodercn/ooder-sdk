3.2 决策引擎核心实现

public class DecisionEngineImpl implements DecisionEngine {
    
    private final LlmProvider llmProvider;
    private final MvelRuleEngine ruleEngine;
    private DecisionMode mode = DecisionMode.ONLINE_FIRST;
    
    private final Cache<String, DecisionResult> decisionCache;
    
    @Override
    public DecisionResult decide(DecisionContext context) {
        // 1. 检查缓存
        String cacheKey = buildCacheKey(context);
        DecisionResult cached = decisionCache.getIfPresent(cacheKey);
        if (cached != null) {
            return cached.withSource(DecisionSource.CACHED);
        }
        
        // 2. 根据模式选择决策方式
        DecisionResult result;
        switch (mode) {
            case ONLINE_ONLY:
                result = decideWithLlm(context);
                break;
            case OFFLINE_ONLY:
                result = decideWithRules(context);
                break;
            case ONLINE_FIRST:
            default:
                result = decideWithFallback(context);
        }
        
        // 3. 缓存结果
        if (result.isSuccess()) {
            decisionCache.put(cacheKey, result);
        }
        
        return result;
    }
    
    private DecisionResult decideWithFallback(DecisionContext context) {
        // 1. 尝试LLM决策
        if (isLlmAvailable()) {
            try {
                return decideWithLlm(context);
            } catch (Exception e) {
                log.warn("LLM决策失败，降级到规则引擎", e);
            }
        }
        
        // 2. 降级到规则引擎
        return decideWithRules(context);
    }
}

3.3 三种决策模式对比

模式 说明 延迟 准确率 适用场景
ONLINE_ONLY 仅使用LLM决策 200-500ms >95% 对准确性要求高，网络稳定
OFFLINE_ONLY 仅使用规则引擎 5-20ms ~80% 离线环境、性能敏感场景
ONLINE_FIRST 优先LLM，失败降级规则 5-500ms >90% 默认模式，兼顾准确性和可用性

4.2 跨层检索实现

public class KnowledgeCapabilityImpl implements KnowledgeCapability {
    
    private final EmbeddingService embeddingService;
    private final VectorStore vectorStore;
    private final KnowledgeBaseService kbService;
    
    @Override
    public KnowledgeResult retrieveCrossLayer(
            String query, 
            KnowledgeLayer startLayer) {
        
        List<RetrievedItem> allItems = new ArrayList<>();
        
        // 向量查询
        float[] queryVector = embeddingService.embed(query);
        
        // 从起始层向下检索
        for (KnowledgeLayer layer : getLayerSequence(startLayer)) {
            Map<String, Object> filters = new HashMap<>();
            filters.put("layer", layer.name());
            
            List<SearchResult> results = vectorStore.search(
                queryVector, topK, filters
            );
            
            // 转换并添加结果
            results.stream()
                .map(r -> toRetrievedItem(r, layer))
                .forEach(allItems::add);
        }
        
        // 按相似度排序并返回
        return KnowledgeResult.builder()
            .items(allItems.stream()
                .sorted(Comparator.comparingDouble(
                    RetrievedItem::getScore).reversed())
                .limit(topK)
                .collect(Collectors.toList()))
            .build();
    }
}

5.3 安全沙箱配置

public class MvelRuleEngineImpl implements MvelRuleEngine {
    
    private final Set<String> allowedClasses = new HashSet<>();
    private final Set<String> deniedMethods = new HashSet<>();
    
    public MvelRuleEngineImpl() {
        // 允许的类
        allowedClasses.add("java.lang.String");
        allowedClasses.add("java.lang.Integer");
        allowedClasses.add("java.util.ArrayList");
        allowedClasses.add("java.util.HashMap");
        allowedClasses.add("java.text.SimpleDateFormat");
        
        // 禁止的方法
        deniedMethods.add("exit");
        deniedMethods.add("exec");
        deniedMethods.add("getRuntime");
        deniedMethods.add("forName");
    }
    
    @Override
    public Object execute(String script, Map<String, Object> context) {
        // 创建沙箱执行器
        ParserContext parserContext = new ParserContext();
        
        // 配置安全限制
        Map<String, Object> variables = new HashMap<>(context);
        variables.put("context", context);
        
        // 执行脚本（带超时）
        return MVEL.executeExpression(
            compileWithCache(script), 
            variables,
            parserContext
        );
    }
}
